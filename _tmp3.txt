export function SensorHistoryPage() { const { token } = useAuth(); const [readings, setReadings] = useState<DashboardData['recentSensorReadings']>([]); const [binCode, setBinCode] = useState(''); const [loading, setLoading] = useState(true); useEffect(() => { const query = binCode ? `?binCode=${encodeURIComponent(binCode)}` : ''; getData<{ readings: DashboardData['recentSensorReadings'] }>(`/api/dashboard/sensor-readings${query}`, token!).then((result) => setReadings(result.readings)).finally(() => setLoading(false)); }, [token, binCode]); return <section className="page"><div className="page-heading"><div><div className="eyebrow">LỊCH SỬ Dữ LIỆU CẢM BIẾN</div><h1>Lịch sử cảm biến</h1><p>Lịch sử dữ liệu cảm biến được lưu trữ tại hệ thống.</p></div></div><label className="wide-search">Lọc theo mã thùng rác<input value={binCode} onChange={(event) => { setLoading(true); setBinCode(event.target.value); }} placeholder="Ví dụ: mã thùng rác đã cấu hình" /></label>{loading ? <div className="empty-state">Đang tải lịch sử cảm biến...</div> : readings.length === 0 ? <div className="empty-state">Không tìm thấy dữ liệu cảm biến nào.</div> : <div className="table-wrap"><table><thead><tr><th>Thùng rác</th><th>Mức độ đầy</th><th>Khoảng cách</th><th>Thời gian</th></tr></thead><tbody>{readings.map((reading) => <tr key={reading.id}><td><strong>{reading.bin.binCode}</strong><span className="muted">{reading.bin.name}</span></td><td>{reading.fillLevel}%</td><td>{reading.distance} cm</td><td>{ago(reading.timestamp)}</td></tr>)}</tbody></table></div>}</section>; }
